"""
Construction & Engineering Proof-of-Work (PoW) Verification & Calculation Suite.
Provides verifiable structural analysis, concrete mix design compliance checks,
and cryptographic PoW attestation for project milestone deliverables.
"""

from dataclasses import dataclass, field
import hashlib
import json
import math
import time
from typing import Any, Dict, List, Optional


@dataclass
class StructuralBeamCheck:
    """Calculates bending moment, shear capacity, and deflection for a simply supported beam."""
    span_m: float  # Length in meters
    dead_load_kn_per_m: float  # Dead load (w_d) in kN/m
    live_load_kn_per_m: float  # Live load (w_l) in kN/m
    elastic_modulus_gpa: float  # Modulus of elasticity (E) in GPa
    moment_of_inertia_cm4: float  # Moment of inertia (I) in cm^4
    yield_strength_mpa: float  # Steel yield strength (f_y) in MPa
    section_modulus_cm3: float  # Elastic section modulus (Z) in cm^3

    def calculate(self) -> Dict[str, Any]:
        # Ultimate limit state design load (Eurocode/AISC factored combination: 1.2D + 1.6L)
        w_u = (1.2 * self.dead_load_kn_per_m) + (1.6 * self.live_load_kn_per_m)
        w_service = self.dead_load_kn_per_m + self.live_load_kn_per_m

        # Max Bending Moment: M_u = (w_u * L^2) / 8 (kNm)
        max_moment_kn_m = (w_u * (self.span_m ** 2)) / 8.0

        # Max Shear Force: V_u = (w_u * L) / 2 (kN)
        max_shear_kn = (w_u * self.span_m) / 2.0

        # Max Deflection at midspan: delta = (5 * w_serv * L^4) / (384 * E * I)
        # Convert units: E from GPa to kN/m2 (1 GPa = 1e6 kN/m2), I from cm4 to m4 (1 cm4 = 1e-8 m4)
        e_kn_m2 = self.elastic_modulus_gpa * 1e6
        i_m4 = self.moment_of_inertia_cm4 * 1e-8
        deflection_m = (5.0 * w_service * (self.span_m ** 4)) / (384.0 * e_kn_m2 * i_m4)
        deflection_mm = deflection_m * 1000.0

        # Allowable deflection limit (L / 360)
        allowable_deflection_mm = (self.span_m * 1000.0) / 360.0

        # Moment resistance: M_rd = (f_y * Z) / gamma_m0 (kNm), gamma_m0 = 1.0
        # f_y in N/mm2 = kN/m2 * 1e-3, Z in cm3 = 1e-6 m3 -> M in kNm
        z_m3 = self.section_modulus_cm3 * 1e-6
        fy_kn_m2 = self.yield_strength_mpa * 1000.0
        moment_resistance_kn_m = fy_kn_m2 * z_m3

        moment_utilization = max_moment_kn_m / moment_resistance_kn_m if moment_resistance_kn_m > 0 else float("inf")
        deflection_pass = deflection_mm <= allowable_deflection_mm
        bending_pass = moment_utilization <= 1.0

        return {
            "factored_load_kn_per_m": round(w_u, 3),
            "max_bending_moment_kn_m": round(max_moment_kn_m, 2),
            "moment_capacity_kn_m": round(moment_resistance_kn_m, 2),
            "moment_utilization": round(moment_utilization, 3),
            "max_shear_kn": round(max_shear_kn, 2),
            "midspan_deflection_mm": round(deflection_mm, 2),
            "allowable_deflection_mm": round(allowable_deflection_mm, 2),
            "bending_check_passed": bending_pass,
            "deflection_check_passed": deflection_pass,
            "status": "PASS" if (bending_pass and deflection_pass) else "FAIL"
        }


@dataclass
class ConcreteMixVerification:
    """Validates structural concrete mix design against target compressive strength (ACI/EN 206)."""
    target_fck_mpa: float
    water_kg: float
    cement_kg: float
    fine_aggregate_kg: float
    coarse_aggregate_kg: float

    def verify(self) -> Dict[str, Any]:
        wc_ratio = self.water_kg / self.cement_kg if self.cement_kg > 0 else float("inf")
        total_density = self.water_kg + self.cement_kg + self.fine_aggregate_kg + self.coarse_aggregate_kg

        # Abram's Law empirical strength estimate: f_c ~ A / B^(w/c)
        # For standard Portland cement Type I:
        estimated_28d_strength = 98.0 / (7.0 ** (1.5 * wc_ratio))

        compliant = estimated_28d_strength >= self.target_fck_mpa and wc_ratio <= 0.50

        return {
            "water_cement_ratio": round(wc_ratio, 3),
            "density_kg_per_m3": round(total_density, 1),
            "target_strength_mpa": self.target_fck_mpa,
            "estimated_28d_strength_mpa": round(estimated_28d_strength, 2),
            "wc_ratio_compliant": wc_ratio <= 0.50,
            "strength_compliant": estimated_28d_strength >= self.target_fck_mpa,
            "status": "PASS" if compliant else "FAIL"
        }


@dataclass
class ProofOfWorkAttestation:
    """Creates a cryptographically verifiable proof-of-work record for project milestones."""
    project_id: str
    subsystem: str
    milestone_name: str
    metrics: Dict[str, Any]
    timestamp: float = field(default_factory=time.time)
    nonce: int = 0
    proof_hash: str = ""

    def generate_proof(self, difficulty: int = 4) -> str:
        """Finds a proof hash with leading zeros according to target difficulty."""
        target_prefix = "0" * difficulty
        payload_base = {
            "project_id": self.project_id,
            "subsystem": self.subsystem,
            "milestone_name": self.milestone_name,
            "metrics": self.metrics,
            "timestamp": self.timestamp,
        }

        nonce = 0
        while True:
            payload_base["nonce"] = nonce
            serialized = json.dumps(payload_base, sort_keys=True).encode("utf-8")
            digest = hashlib.sha256(serialized).hexdigest()
            if digest.startswith(target_prefix):
                self.nonce = nonce
                self.proof_hash = digest
                return digest
            nonce += 1


def run_build_proof():
    # 1. Structural Engineering Analysis: Steel Girder W310x38.7
    girder = StructuralBeamCheck(
        span_m=6.5,
        dead_load_kn_per_m=8.5,
        live_load_kn_per_m=12.0,
        elastic_modulus_gpa=200.0,
        moment_of_inertia_cm4=8490.0,
        yield_strength_mpa=355.0,
        section_modulus_cm3=549.0
    )
    beam_results = girder.calculate()

    # 2. Materials Engineering Analysis: C30/37 High Performance Foundation Mix
    concrete = ConcreteMixVerification(
        target_fck_mpa=30.0,
        water_kg=175.0,
        cement_kg=390.0,
        fine_aggregate_kg=780.0,
        coarse_aggregate_kg=1055.0
    )
    concrete_results = concrete.verify()

    # 3. Cryptographic Proof-of-Work Attestation
    pow_record = ProofOfWorkAttestation(
        project_id="1593928-SN-CE",
        subsystem="Construction_and_Engineering",
        milestone_name="Show Your Build Weekly #5: Proof-of-Work",
        metrics={
            "beam_check": beam_results,
            "concrete_verification": concrete_results
        }
    )
    proof_hash = pow_record.generate_proof(difficulty=4)

    output = {
        "bounty_id": "1593928",
        "category": "Construction_and_Engineering",
        "milestone": "Weekly Build #5 PoW",
        "pow_hash": proof_hash,
        "nonce": pow_record.nonce,
        "structural_analysis": beam_results,
        "materials_verification": concrete_results,
        "attestation_status": "VERIFIED"
    }

    print(json.dumps(output, indent=2))
    return output


if __name__ == "__main__":
    run_build_proof()

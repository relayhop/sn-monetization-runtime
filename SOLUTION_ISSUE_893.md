def analyze_sequence_termination(start_val: int, transition_fn, max_steps: int = 100000):
    """
    Analyzes whether a sequence generated by transition_fn starting at start_val
    terminates in a loop, returning cycle length and path history.
    """
    slow = start_val
    fast = start_val
    history = [start_val]
    steps = 0

    # Phase 1: Meeting point using Floyd's cycle-finding
    while steps < max_steps:
        slow = transition_fn(slow)
        fast = transition_fn(transition_fn(fast))
        steps += 1
        history.append(slow)
        if slow == fast:
            break
    else:
        return {"terminates_in_loop": False, "reason": "Exceeded max steps without cycle"}

    # Phase 2: Find cycle start and length
    cycle_start = start_val
    while cycle_start != slow:
        cycle_start = transition_fn(cycle_start)
        slow = transition_fn(slow)

    current = transition_fn(cycle_start)
    cycle_length = 1
    while current != cycle_start:
        current = transition_fn(current)
        cycle_length += 1

    return {
        "terminates_in_loop": True,
        "cycle_start": cycle_start,
        "cycle_length": cycle_length,
        "steps_to_detect": steps
    }
